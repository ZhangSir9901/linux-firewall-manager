# 一键下载并运行防火墙管理脚本：
bash <(curl -s -L https://raw.githubusercontent.com/YOUR_USERNAME/linux-firewall-manager/main/firewall_manager.sh)

# 或者用 wget 的方式：
wget -O firewall_manager.sh https://raw.githubusercontent.com/YOUR_USERNAME/linux-firewall-manager/main/firewall_manager.sh && bash firewall_manager.sh
